% =====================================================================
% ATS-OPTIMIZED RESUME TEMPLATE
% Built for maximum parsing reliability across Workday, Greenhouse,
% Lever, Taleo, iCIMS, and similar Applicant Tracking Systems.
%
% RULES FOLLOWED IN THIS TEMPLATE:
% - Single column, no text boxes, no tables-as-layout
% - Standard section names (Summary, Experience, Skills, Projects,
%   Education, Certifications) -- never renamed creatively
% - Plain bullets only (no icons, stars, or rating graphics)
% - Consistent Job Title / Company / Location / Dates pattern
% - Standard, widely-embedded font (Times, via mathptmx) --
%   no exotic fonts that can break PDF text-extraction
% - No headers/footers carrying essential content
% - No columns, no colored text, no images
% =====================================================================

\documentclass[10.5pt, letterpaper]{article}
\usepackage[letterpaper, top=0.6in, bottom=0.6in, left=0.75in, right=0.75in]{geometry}
\usepackage{mathptmx}          % Times-family font -- safe, universal, embeds cleanly
\usepackage[T1]{fontenc}
\usepackage{enumitem}
\usepackage{hyperref}
\usepackage{titlesec}
% Removed ragged2e to fix compilation

\pagestyle{empty}              % no headers/footers -- nothing for ATS to mis-parse

\hypersetup{
  colorlinks=true,
  linkcolor=black,
  urlcolor=black,
  pdfborder={0 0 0}
}

% ---- Section formatting: plain, bold, uppercase, ruled underline ----
\titleformat{\section}{\bfseries\large}{}{0pt}{}[\titlerule]
\titlespacing{\section}{0pt}{10pt}{6pt}

% ---- Entry pattern: fully sequential, left-to-right, NO \hfill ----
% \hfill-based right-alignment can cause PDF text-extraction to reorder
% dates/locations out of sequence in some ATS parsers (verified via
% pdftotext raw-order test). This format guarantees correct reading
% order for every parser, at a small cost to visual right-alignment.
\newcommand{\entry}[4]{%
  \noindent\textbf{#1} --- #3, #4 \textbar\ #2\par
  \vspace{2pt}
}

% ---- Simplified entry for Projects (no company/location) ----
\newcommand{\projectentry}[2]{%
  \noindent\textbf{#1} \textbar\ #2\par
  \vspace{2pt}
}

% ---- Plain bullet list, no custom symbols ----
\setlist[itemize]{label=\textbullet, leftmargin=16pt, itemsep=1pt, topsep=2pt, parsep=0pt}

\newcommand{\resumeItem}[1]{\item #1}
